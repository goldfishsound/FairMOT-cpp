cmake_minimum_required(VERSION 3.5 FATAL_ERROR)

find_package(GTest REQUIRED)
add_executable(my_test my_test.cpp)

message(STATUS "Package directory: ${PACKAGE_DIR}")
target_include_directories(my_test
    PRIVATE
        "{PACKAGE_DIR}/include"
)

# Apple Silicon specific flags
if(APPLE)
    find_library(ACCELERATE_FRAMEWORK
        NAMES Accelerate
        PATHS ${CMAKE_SYSTEM_FRAMEWORK_PATH}
        PATH_SUFFIXES Frameworks
    )
    # Check if the architecture is arm64 (Apple Silicon)
    if(CMAKE_SYSTEM_PROCESSOR STREQUAL "arm64")
        set(CMAKE_OSX_ARCHITECTURES "arm64")
    else()
        set(CMAKE_OSX_ARCHITECTURES "x86_64")
    endif()
endif()

# Find all static and dynamic libraries under the dependencies directory
file(GLOB_RECURSE DEP_LIBS
    "${PACKAGE_DIR}/lib/dependencies/*.a"
    "${PACKAGE_DIR}/lib/dependencies/*.dylib"
)

target_link_libraries(my_test 
    "${PACKAGE_DIR}/lib/libAWTracker.dylib"
    ${DEP_LIBS}
    ${ACCELERATE_FRAMEWORK}
    GTest::gtest 
    GTest::gtest_main
    )

# Set rpath so dynamic libraries under dependencies are found at runtime
set_target_properties(my_test PROPERTIES
    BUILD_RPATH "${PACKAGE_DIR}/lib/dependencies"
    INSTALL_RPATH "${PACKAGE_DIR}/lib/dependencies"
)

add_test(NAME MyTest COMMAND my_test)
