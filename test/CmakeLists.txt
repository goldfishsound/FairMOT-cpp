cmake_minimum_required(VERSION 3.5 FATAL_ERROR)

set(GTEST_PATH "/opt/homebrew/Cellar/googletest/1.17.0/lib/cmake/GTest")
find_package(GTest REQUIRED PATHS ${GTEST_PATH} NO_DEFAULT_PATH)

    PRIVATE
        ${API_SOURCES}
        ${PROJECT_SOURCE_DIR}/src
        ${PROJECT_SOURCE_DIR}/third_party/mini
        ${GTEST_INCLUDE_DIRS}
)



set(LIBS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../build")

    GTest::gtest_main 
    Eigen3::Eigen
    ${OpenCV_LIBRARIES}
    ${Python3_LIBRARIES}
    ${TORCH_LIBRARIES}
    ${VISION_LIBRARY}
    ${CXX_LIB}
    ${OMP_LIBRARY}
    ${ACCELERATE_FRAMEWORK}
    ${LIBS_DIR}/libfairmotlib.a)
